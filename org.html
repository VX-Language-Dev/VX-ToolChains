// 代数类型与模式匹配
type Shape = Circle(f64) | Rect(f64, f64)

fn area(s Shape) f64 {
    match s {
        Circle(r)  => 3.14159 * r * r,
        Rect(w, h) => w * h,
    }
}

fn main() {
    shapes := [Circle(2.0), Rect(3.0, 4.0)]
    total  := shapes.map(area).sum()
    println("总面积: ${total}")
}
// 结构化并发 · 编译期保证无数据竞争
spawn fn fetch(url string) Result<string> {
    return http.get(url)?.text()
}

fn main() {
    urls := ["vx.dev", "pkg.vx.dev", "blog.vx.dev"]

    results := urls
        .map(spawn fetch(it))
        .await_all()

    println("抓取完成: ${results.len()} 个站点")
}
